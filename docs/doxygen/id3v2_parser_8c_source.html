46uint32_t id3v2ParseExtendedTagHeader(uint8_t *in, size_t inl, uint8_t version,
47 Id3v2ExtendedTagHeader **extendedTagHeader) {
48 if (in == NULL || inl == 0) {
49 *extendedTagHeader = NULL;
50 return 0;
51 }
52
53 ByteStream *stream = byteStreamCreate(in, inl);
54
55 if (!stream) {
56 *extendedTagHeader = NULL;
57 return 0;
58 }
59
60 // values needed
61 uint32_t padding = 0;
62 uint32_t crc = 0;
63 bool update = false;
64 bool tagRestrictions = false;
65 uint8_t restrictions = 0;
66
67 // values for parsing
68 size_t offset = 0;
69 unsigned char crcBytes[5] = {0, 0, 0, 0, 0};
70 bool hasCrc = false;
71 bool hasRestrictions = false;
72
73 // size return
74 uint32_t hSize = 0;
75
76 if (version == ID3V2_TAG_VERSION_3 || version == ID3V2_TAG_VERSION_4) {
77 hSize = byteStreamReturnInt(stream);
78 if (!hSize) {
79 *extendedTagHeader = NULL;
80 byteStreamDestroy(stream);
81 return 0;
82 }
83
84 if (hSize > (stream->bufferSize - stream->cursor)) {
85 offset = hSize - (stream->bufferSize - stream->cursor);
86 }
87 }
88
89 ByteStream *innerStream = byteStreamCreate(byteStreamCursor(stream), hSize - offset);
90
91 switch (version) {
92 case ID3V2_TAG_VERSION_3: {
93 const int8_t flags = (int8_t) byteStreamReadBit(innerStream, 7);
94
95 // read flags
96 if (flags == -1) {
97 break;
98 }
99
100 // skip over flags
101 if (!byteStreamSeek(innerStream, 2, SEEK_CUR)) {
102 break;
103 }
104
105 // read padding
106 padding = byteStreamReturnInt(innerStream);
107
108 if (!padding && flags == 0) {
109 break;
110 }
111
112 // check to see if a crc is there
113 if (flags) {
114 crc = byteStreamReturnInt(innerStream);
115 if (!crc) {
116 break;
117 }
118 }
119
120 break;
121 }
122 case ID3V2_TAG_VERSION_4:
123
124 // number of flag bytes
125 if (!byteStreamSeek(innerStream, 1, SEEK_CUR)) {
126 break;
127 }
128
129 // read flags
130 if (byteStreamReadBit(innerStream, 6)) {
131 update = true;
132 }
133
134 if (byteStreamReadBit(innerStream, 5)) {
135 hasCrc = true;
136 }
137
138 if (byteStreamReadBit(innerStream, 4)) {
139 hasRestrictions = true;
140 }
141
142 if (!byteStreamSeek(innerStream, 1, SEEK_CUR)) {
143 break;
144 }
145
146 // read crc
147 if (hasCrc) {
148 if (!byteStreamRead(innerStream, (uint8_t *) crcBytes, 5)) {
149 break;
150 }
151
152 crc = byteSyncintDecode(btoi(crcBytes, 5));
153 }
154
155 // read restrictions
156 if (hasRestrictions) {
157 for (uint8_t k = 8; k > 0; k--) {
158 int bit = byteStreamReadBit(innerStream, k - 1);
159
160 if (bit == -1) {
161 break;
162 }
163
164 restrictions = setBit(restrictions, k - 1, (bool) bit);
165 }
166
167 tagRestrictions = true;
168
169 if (!byteStreamSeek(innerStream, 1, SEEK_CUR)) {
170 break;
171 }
172 }
173
174 break;
175
176 // no support
177 default:
178 byteStreamDestroy(innerStream);
179 *extendedTagHeader = NULL;
180 return 0;
181 }
182
183 // clean up
184 byteStreamDestroy(stream);
185 *extendedTagHeader = id3v2CreateExtendedTagHeader(padding, crc, update, tagRestrictions, restrictions);
186 uint32_t walk = innerStream->cursor + 4;
187
188 if (innerStream != NULL) {
189 byteStreamDestroy(innerStream);
190 }
191
192 return walk;
193}

602static size_t internal_id3v2ParseFrameEncodedStringContent(ByteStream *innerStream, List *context, List *entries,
603 Id3v2ContentContext *cc, size_t upperBound, uint8_t **data) {
604 *data = NULL;
605 size_t dataSize = 0;
606
607 // Find the position of the "encoding" key in the context list,
608 // skipping iter_context entries as they don't map to real content entries.
609 size_t poscc = 0;
610 ListIter contentContextIter = listCreateIterator(context);
611 void *tmp = NULL;
612 while ((tmp = listIteratorNext(&contentContextIter)) != NULL) {
613 if (((Id3v2ContentContext *) tmp)->type == iter_context) {
614 poscc--;
615 }
616 if (((Id3v2ContentContext *) tmp)->key == id3v2djb2("encoding")) {
617 break;
618 }
619 poscc++;
620 }
621
622 // Walk the already-parsed entries to find the encoding byte at poscc.
623 uint8_t encoding = 0;
624 size_t posce = 0;
625 ListIter contentEntryIter = listCreateIterator(entries);
626 while ((tmp = listIteratorNext(&contentEntryIter)) != NULL) {
627 if (poscc == posce) {
628 encoding = ((uint8_t *) ((Id3v2ContentEntry *) tmp)->entry)[0];
629 }
630 posce++;
631 }
632
633 // Read string bytes according to encoding.
634 switch (encoding) {
635 case ID3V2_ENCODING_ISO_8859_1:
636 case ID3V2_ENCODING_ASCII:
637 case ID3V2_ENCODING_UTF8:
638 *data = byteStreamReturnUtf8(innerStream, &dataSize);
639 if (*data == NULL && dataSize == 0) {
640 byteStreamSeek(innerStream, 1, SEEK_CUR);
641 *data = calloc(sizeof(unsigned char), 1);
642 if (*data == NULL) { return 0; }
643 dataSize = 1;
644 }
645 break;
646 case ID3V2_ENCODING_UTF16BE:
647 case ID3V2_ENCODING_UTF16LE:
648 *data = byteStreamReturnUtf16(innerStream, &dataSize);
649 if (*data == NULL && dataSize == 0) {
650 byteStreamSeek(innerStream, 2, SEEK_CUR);
651 *data = calloc(sizeof(unsigned char), 2);
652 if (*data == NULL) { return 0; }
653 dataSize = 2;
654 }
655 break;
656 default:
657 break;
658 }
659
660 // Clamp to [cc->min, cc->max].
661 if (dataSize > cc->max) {
662 uint8_t *reallocPtr = realloc(*data, cc->max);
663 if (reallocPtr == NULL) {
664 free(*data);
665 *data = NULL;
666 return 0;
667 }
668 dataSize = cc->max;
669 *data = reallocPtr;
670 } else if (cc->min > dataSize) {
671 uint8_t *reallocPtr = realloc(*data, cc->min);
672 if (reallocPtr == NULL) {
673 free(*data);
674 *data = NULL;
675 return 0;
676 }
677 memset(reallocPtr + dataSize, 0, cc->min - dataSize);
678 *data = reallocPtr;
679 dataSize = cc->min;
680 } else if (dataSize > upperBound) {
681 dataSize = upperBound + 1;
682 uint8_t *reallocPtr = realloc(*data, dataSize);
683 if (reallocPtr == NULL) {
684 free(*data);
685 return 0;
686 }
687 *data = reallocPtr;
688 (*data)[dataSize - 1] = 0;
689 }
690
691 return dataSize;
692}

733static void internal_copyNBits(const unsigned char *src, unsigned char *dest,
734 int startBit, int nBits) {
735 if (!src || nBits <= 0) return;
736
737 for (int i = 0; i < nBits; i++) {
738 int srcBitPos = startBit + i;
739 int srcByte = srcBitPos / CHAR_BIT;
740 int srcOff = (CHAR_BIT - 1) - (srcBitPos % CHAR_BIT); // MSB-first from src
741
742 // pack so result is little-endian readable: last bits go in dest[0]
743 int destBitPos = (nBits - 1) - i;
744 int destByte = destBitPos / CHAR_BIT;
745 int destOff = destBitPos % CHAR_BIT; // LSB-first within byte
746
747 unsigned char bit = (src[srcByte] >> srcOff) & 0x01;
748 dest[destByte] |= (bit << destOff);
749 }
750}

766static size_t internal_id3v2ParseFrameBitContent(ByteStream *innerStream, const Id3v2ContentContext *cc,
767 const ListIter *iter, size_t *concurrentBitCount,
768 uint32_t *expectedContentSize, uint8_t **dataOut) {
769
770 if (!innerStream || !cc || !iter || !concurrentBitCount || !expectedContentSize || !dataOut) {
771 return 0;
772 }
773
774 *dataOut = NULL;
775
776 size_t nBits = (cc->min >= cc->max) ? cc->min : cc->max;
777 size_t dataSize = (nBits + (CHAR_BIT - 1)) / CHAR_BIT;
778
779 uint8_t *data = calloc(dataSize, 1);
780 if (!data) {
781 return 0;
782 }
783
784 // use concurrentBitCount to find both which byte and which bit to start from
785 size_t startByte = *concurrentBitCount / CHAR_BIT;
786 size_t startBit = *concurrentBitCount % CHAR_BIT;
787
788 internal_copyNBits(byteStreamCursor(innerStream) + startByte, data, (int)startBit, (int)nBits);
789 // reverse the byte order
790 for (size_t i = 0; i < dataSize / 2; i++) {
791 uint8_t tmp = data[i];
792 data[i] = data[dataSize - 1 - i];
793 data[dataSize - 1 - i] = tmp;
794 }
795
796 *concurrentBitCount += nBits;
797
798 // peek at next context
799 Id3v2ContextType nextType = unknown_context;
800 if (iter->current && iter->current->data) {
801 nextType = ((Id3v2ContentContext *)iter->current->data)->type;
802 }
803
804 // only advance stream when the bit run ends
805 if (nextType != bit_context) {
806 size_t flushBytes = *concurrentBitCount / CHAR_BIT;
807 if (flushBytes) {
808 byteStreamSeek(innerStream, (long)flushBytes, SEEK_CUR);
809 *expectedContentSize = (*expectedContentSize < flushBytes) ? 0 : *expectedContentSize - flushBytes;
810 }
811 *concurrentBitCount = 0;
812 }
813
814 *dataOut = data;
815 return dataSize;
816}

850 *frame = NULL;
851 return 0;
852 }
853
854 ByteStream *stream = byteStreamCreate(in, inl);
855
856 if (!stream || !context) {
857 *frame = NULL;
858 return 0;
859 }
860
861 // values needed
862 Id3v2FrameHeader *header = NULL;
863 List *entries = NULL;
864 *frame = NULL;
865
866 // needed for parsing
867 size_t walk = 0;
868 size_t currIterations = 0;
869 size_t concurrentBitCount = 0;
870 uint32_t expectedHeaderSize = 0;
871 uint32_t expectedContentSize = 0;
872 ByteStream *innerStream = NULL;
873 ListIter iter;
874 ListIter iterStorage;
875 void *contextData = NULL;
876
877 expectedHeaderSize = id3v2ParseFrameHeader(stream->buffer, stream->bufferSize, version, &header,
878 &expectedContentSize);
879 walk += expectedHeaderSize;
880 if (!expectedHeaderSize) {
881 if (header != NULL) {
882 id3v2DestroyFrameHeader(&header);
883 }
884
885 byteStreamDestroy(stream);
886 return 0;

890 // just in case both args are not null
891 if (header != NULL) {
892 id3v2DestroyFrameHeader(&header);
893 }

903 id3v2CopyContentEntry);

906 // but parse the content from the resynchronised bytes
907 const uint32_t storedContentSize = expectedContentSize;
908 const bool frameUnsync = (version == ID3V2_TAG_VERSION_4 && header->unsynchronisation);
909
910 if (frameUnsync && innerStream != NULL) {
911 innerStream->bufferSize = id3v2RemoveUnsynchronisation(innerStream->buffer, innerStream->bufferSize);
912 expectedContentSize = (uint32_t) innerStream->bufferSize;
913 }

915
916 // is a frame compressed or encrypted?
917 // if so a generic context will be used meaning it's not up to me to decompress or unencrypted + if it's a text frame none
918 // of the reads or writes will work until someone reparses the data after its changed.
919 if (header->encryptionSymbol > 0 || header->decompressionSize > 0) {
920 uint8_t *data = NULL;
921 size_t dataSize = 0;
922 List *gContext = id3v2CreateGenericFrameContext();
923 Id3v2ContentContext *cc = (Id3v2ContentContext *) gContext->head->data;
924
925 if (cc->min >= cc->max) {
926 // no trust
927 dataSize = cc->min;
928 } else {
929 dataSize = cc->max;
930 }
931
932 if (dataSize > expectedContentSize) {
933 dataSize = expectedContentSize;
934 }
935
936 data = malloc(dataSize);
937
938 if (!byteStreamRead(innerStream, data, dataSize)) {
939 memset(data, 0, dataSize);
940 }
941
942 listInsertBack(entries, id3v2CreateContentEntry(data, dataSize));
943 free(data);
944
945 walk += frameUnsync ? storedContentSize : innerStream->cursor;
946 *frame = id3v2CreateFrame(header, gContext, entries);
947 byteStreamDestroy(innerStream);
948 byteStreamDestroy(stream);
949 return walk;
950 }
951
952 iter = listCreateIterator(context);
953
954 while ((contextData = listIteratorNext(&iter)) != NULL) {
955 Id3v2ContentContext *cc = (Id3v2ContentContext *) contextData;
956
957 switch (cc->type) {
958 // encoded strings
959 case encodedString_context: {
960 uint8_t *data = NULL;
961 size_t dataSize = 0;
962
963 dataSize = internal_id3v2ParseFrameEncodedStringContent(innerStream, context, entries, cc,
964 expectedContentSize, &data);
965
966 if (data == NULL) {
967 listFree(entries);
968 byteStreamDestroy(innerStream);
969 byteStreamDestroy(stream);
970 id3v2DestroyFrameHeader(&header);
971 *frame = NULL;
972 return 0;
973 }
974
975 listInsertBack(entries, id3v2CreateContentEntry(data, dataSize));
976 free(data);
977
978 expectedContentSize = ((expectedContentSize < dataSize) ? 0 : expectedContentSize - dataSize);
979 }
980 break;
981 // only characters found within the latin1 character set
982 case latin1Encoding_context: {
983 uint8_t *data = NULL;
984 size_t dataSize = 0;
985
986 dataSize = internal_id3v2ParseFrameLatin1Content(innerStream, cc, expectedContentSize, &data);
987
988 // parsing or memmory error
989 if (data == NULL && dataSize == 0) {
990 free(data);
991 listFree(entries);
992 byteStreamDestroy(innerStream);
993 byteStreamDestroy(stream);
994 id3v2DestroyFrameHeader(&header);
995 *frame = NULL;
996 return 0;
997 }
998
999 listInsertBack(entries, (void *) id3v2CreateContentEntry(data, dataSize));
1000 free(data);
1001
1002 expectedContentSize = ((expectedContentSize < dataSize) ? 0 : expectedContentSize - dataSize);
1003 break;
1004 }
1005 // numbers (handled the same way)
1006 case binary_context:
1007 case noEncoding_context:
1008 case precision_context:
1009 case numeric_context: {
1010 uint8_t *data = NULL;
1011 size_t dataSize = 0;
1012 dataSize = internal_id3v2ParseFrameBlobContent(innerStream, cc, expectedContentSize, &data);
1013
1014 listInsertBack(entries, id3v2CreateContentEntry(data, dataSize));
1015 free(data);
1016
1017 expectedContentSize = ((expectedContentSize < dataSize) ? 0 : expectedContentSize - dataSize);
1018 }
1019 break;
1020 case bit_context: {
1021 uint8_t *data = NULL;
1022 size_t dataSize = internal_id3v2ParseFrameBitContent(
1023 innerStream, cc, &iter,
1024 &concurrentBitCount,
1025 &expectedContentSize,
1026 &data);
1027 if (!data) {
1028 listFree(entries);
1029 byteStreamDestroy(innerStream);
1030 byteStreamDestroy(stream);
1031 id3v2DestroyFrameHeader(&header);
1032 *frame = NULL;
1033 return 0;
1034 }
1035
1036 listInsertBack(entries, id3v2CreateContentEntry(data, dataSize));
1037 free(data);
1038 }
1039 break;
1040 case iter_context: {
1041 if (!currIterations) {
1042 iterStorage = iter;
1043
1044 iter = listCreateIterator(context);
1045
1046 for (size_t i = 0; i < cc->min; i++) {
1047 listIteratorNext(&iter);
1048 }
1049 }
1050
1051 if (currIterations != cc->max && currIterations != 0) {
1052 iter = listCreateIterator(context);
1053
1054 for (size_t i = 0; i < cc->min; i++) {
1055 listIteratorNext(&iter);
1056 }
1057 }
1058
1059
1060 if (currIterations >= cc->max) {
1061 iter = iterStorage;
1062
1063 for (size_t i = 0; i < currIterations; i++) {
1064 listIteratorNext(&iter);
1065 }
1066
1067 currIterations = 0;
1068 }
1069
1070 currIterations++;
1071 }
1072 break;
1073 case adjustment_context: {
1074
1075 size_t dataSize = 0;
1076 uint8_t *data = NULL;
1077 dataSize = internal_id3v2ParseFrameAdjustmentContent(innerStream, context, entries, cc,
1078 expectedContentSize, &data);
1079 listInsertBack(entries, id3v2CreateContentEntry(data, dataSize));
1080 free(data);
1081
1082 expectedContentSize = ((expectedContentSize < dataSize) ? 0 : expectedContentSize - dataSize);
1083 }
1084 break;
1085 // no support
1086 case unknown_context:
1087 default:
1088 byteStreamSeek(innerStream, 0, SEEK_END);
1089 break;
1090 }
1091
1092 if (expectedContentSize == 0 || byteStreamGetCh(innerStream) == EOF) {
1093 break;
1094 }
1095 }
1096
1097
1098 // enforce frame size from header parsing
1099 walk += frameUnsync ? storedContentSize : innerStream->bufferSize;
1100
1101 *frame = id3v2CreateFrame(header, listDeepCopy(context), entries);
1102 byteStreamDestroy(innerStream);
1103 byteStreamDestroy(stream);
1104 return walk;
1105}

1135Id3v2Tag *id3v2ParseTagFromBuffer(uint8_t *in, size_t inl, HashTable *userPairs) {
1136 if (in == NULL || inl == 0) {
1137 return NULL;
1138 }
1139
1140 ByteStream *stream = byteStreamCreate(in, inl);
1141
1142 if (!stream) {
1143 return NULL;
1144 }
1145
1146 bool exit = false;
1147 uint32_t read = 0;
1148 uint32_t tagSize = 0;
1149 uint8_t id[ID3V2_TAG_ID_SIZE];
1150 Id3v2TagHeader *header = NULL;
1151 Id3v2ExtendedTagHeader *ext = NULL;
1152 List *frames = listCreate(id3v2PrintFrame, id3v2DeleteFrame, id3v2CompareFrame, id3v2CopyFrame);
1153
1154 // locate the start of the tag
1155 while (true) {
1156 if (!byteStreamRead(stream, id, ID3V2_TAG_ID_SIZE)) {
1157 listFree(frames);
1158 byteStreamDestroy(stream);
1159 return NULL;
1160 }
1161
1162 if (btoi(id, ID3V2_TAG_ID_SIZE) == ID3V2_TAG_ID_MAGIC_NUMBER_H) {
1163 stream->cursor = stream->cursor - ID3V2_TAG_ID_SIZE;
1164 break;
1165 }
1166 }
1167
1168 while (true) {
1169 read = id3v2ParseTagHeader(byteStreamCursor(stream), stream->bufferSize - stream->cursor, &header, &tagSize);
1170
1171 if (read == 0 || header == NULL || tagSize == 0) {
1172 break;
1173 }
1174
1175 if (!byteStreamSeek(stream, read, SEEK_CUR)) {
1176 break;
1177 }
1178
1179 // shorten to exclude none tag data
1180 byteStreamResize(stream, stream->cursor + tagSize);
1181
1182 // v2.4 unsynchronisation is applied per frame and frame sizes describe the stored bytes,
1183 // so only v2.2/v2.3 undo it over the whole tag body
1184 if (id3v2ReadUnsynchronisationIndicator(header) == 1 &&
1185 (header->majorVersion == ID3V2_TAG_VERSION_2 || header->majorVersion == ID3V2_TAG_VERSION_3)) {
1186 size_t bodySize = stream->bufferSize - stream->cursor;
1187 size_t decodedSize = id3v2RemoveUnsynchronisation(byteStreamCursor(stream), bodySize);
1188
1189 stream->bufferSize = stream->cursor + decodedSize;
1190 tagSize = (uint32_t) decodedSize;
1191 }
1192
1193 if ((header->majorVersion == ID3V2_TAG_VERSION_3 || header->majorVersion == ID3V2_TAG_VERSION_4) &&
1194 id3v2ReadExtendedHeaderIndicator(header) == 1) {
1195 read = id3v2ParseExtendedTagHeader(byteStreamCursor(stream), stream->bufferSize - stream->cursor,
1196 header->majorVersion, &ext);
1197 header->extendedHeader = ext;
1198
1199 if (read == 0 || ext == NULL) {
1200 break;
1201 }
1202
1203 tagSize = ((tagSize < read) ? 0 : tagSize - read);
1204 byteStreamSeek(stream, read, SEEK_CUR);
1205 }
1206
1207 HashTable *pairs = id3v2CreateDefaultIdentifierContextPairings(header->majorVersion);
1208
1209 while (tagSize) {
1210 Id3v2Frame *frame = NULL;
1211 uint8_t frameId[ID3V2_FRAME_ID_MAX_SIZE + 1] = {0};
1212
1213 if (header->majorVersion == ID3V2_TAG_VERSION_3 || header->majorVersion == ID3V2_TAG_VERSION_4) {
1214 if (!byteStreamRead(stream, frameId, ID3V2_FRAME_ID_MAX_SIZE)) {
1215 exit = true;
1216 break;
1217 }
1218 frameId[ID3V2_FRAME_ID_MAX_SIZE] = '\0';
1219
1220 stream->cursor = stream->cursor - ID3V2_FRAME_ID_MAX_SIZE;
1221 } else if (header->majorVersion == ID3V2_TAG_VERSION_2) {
1222 if (!byteStreamRead(stream, frameId, ID3V2_FRAME_ID_MAX_SIZE - 1)) {
1223 exit = true;
1224 break;
1225 }
1226 frameId[ID3V2_FRAME_ID_MAX_SIZE - 1] = '\0';
1227
1228 stream->cursor = stream->cursor - (ID3V2_FRAME_ID_MAX_SIZE - 1);
1229 }
1230
1231 // check default pairings (pass 1/4)
1232 List *context = hashTableRetrieve(pairs, (char *) frameId);
1233
1234 // check user supplied ones (pass 2/4)
1235 if (context == NULL) {
1236 context = hashTableRetrieve(userPairs, (char *) frameId);
1237 }
1238
1239 // special considerations (pass 3/4)
1240 if (context == NULL) {
1241 if (frameId[0] == 'T') {
1242 context = hashTableRetrieve(pairs, "T");
1243 }
1244
1245 if (frameId[0] == 'W') {
1246 context = hashTableRetrieve(pairs, "W");
1247 }
1248 }
1249
1250 // use generic (pass 4/4)
1251 if (context == NULL) {
1252 context = hashTableRetrieve(pairs, "?");
1253 }
1254
1255 read = id3v2ParseFrame(byteStreamCursor(stream), stream->bufferSize - stream->cursor, context,
1256 header->majorVersion, &frame);
1257
1258 if (read == 0 || frame == NULL) {
1259 exit = true;
1260 break;
1261 }
1262
1263 listInsertBack(frames, frame);
1264 tagSize = ((tagSize < read) ? 0 : tagSize - read);
1265 byteStreamSeek(stream, read, SEEK_CUR);
1266 }
1267
1268 if (pairs != NULL) {
1269 hashTableFree(pairs);
1270 }
1271
1272 // double loop break
1273 if (exit) {
1274 break;
1275 }
1276
1277 byteStreamDestroy(stream);
1278 return id3v2CreateTag(header, frames);
1279 }
1280
1281 byteStreamDestroy(stream);
1282 return id3v2CreateTag(header, frames);
1283}
